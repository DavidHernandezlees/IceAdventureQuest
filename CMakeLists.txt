cmake_minimum_required(VERSION 3.22)
project(IceAdventureQuest LANGUAGES C CXX)
set(COCOS2DX_ROOT_PATH "${CMAKE_CURRENT_SOURCE_DIR}/engine")
list(APPEND CMAKE_MODULE_PATH "${COCOS2DX_ROOT_PATH}/cmake/Modules")
include(CocosBuildSet)
add_subdirectory("${COCOS2DX_ROOT_PATH}/cocos" "${ENGINE_BINARY_PATH}/cocos/core")
if(ANDROID)
    target_precompile_headers(cocos2d PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:${CMAKE_CURRENT_SOURCE_DIR}/cmake/EngineHeaders.h>")
endif()
set(GAME_SOURCES Classes/AppDelegate.cpp Classes/Campaign.cpp Classes/Puzzle.cpp Classes/OriginalBoard.cpp Classes/OriginalLevel.cpp Classes/LevelRules.cpp
    Classes/GameScene.cpp Classes/MainMenuScene.cpp Classes/LoadingScene.cpp Classes/Interface.cpp)
if(ANDROID)
    set(GAME_TARGET iaqgame)
    add_subdirectory("${COCOS2DX_ROOT_PATH}/cocos/platform/android" "${ENGINE_BINARY_PATH}/cocos/platform")
    add_library(iaqgame SHARED ${GAME_SOURCES} proj.android/app/src/main/cpp/main.cpp)
    target_link_libraries(iaqgame -Wl,--whole-archive cpp_android_spec -Wl,--no-whole-archive)
elseif(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "Configure with -A Win32 for the bundled Cocos2d-x dependencies.")
    endif()
    set(GAME_TARGET IceAdventureQuest)
    add_executable(IceAdventureQuest WIN32 ${GAME_SOURCES} proj.win32/main.cpp)
else()
    message(FATAL_ERROR "Choose the Android NDK toolchain or a 32-bit Windows MSVC generator.")
endif()
target_include_directories(${GAME_TARGET} PRIVATE Classes)
target_link_libraries(${GAME_TARGET} cocos2d)
setup_cocos_app_config(${GAME_TARGET})
if(WIN32)
    cocos_copy_target_dll(${GAME_TARGET})
    add_custom_target(game_assets
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Resources" "${CMAKE_BINARY_DIR}/bin/${GAME_TARGET}/$<CONFIG>/Resources"
        VERBATIM)
    add_dependencies(${GAME_TARGET} game_assets)
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT ${GAME_TARGET})
endif()
